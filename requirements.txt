# MyFileLab — dependencies
# Install: pip install -r requirements.txt

# UI
PySide6

# PDF tools
PyMuPDF          # Compress, MD→PDF, OCR rendering
pypdf            # Merge, Split, Delete pages, OCR PDF merge
pdf2docx         # PDF → Word
pdfplumber       # PDF → Excel (table extraction)
openpyxl         # PDF → Excel (write .xlsx)
markdown         # MD → PDF parsing
cryptography     # Encrypted PDF support (used by pypdf)
# OCR: uses Tesseract binary directly (subprocess) — install from:
# https://github.com/UB-Mannheim/tesseract/wiki

# Image tools
Pillow           # Enhance, Batch Export

# Video & Audio downloader
yt-dlp           # Download from 1000+ platforms
imageio-ffmpeg   # Bundled ffmpeg — no system install needed
